} function doPost(e) { const lock = LockService.getScriptLock(); try { lock.waitLock(10000); setupDatabase(); // Zero-config auto DB setup let requestData = {}; if (e && e.postData && e.postData.contents) { requestData = JSON.parse(e.postData.contents); } const action = requestData.action; let responseData = { success: true }; if (action === 'getInitialData') { responseData = getInitialData(); } else if (action === 'saveSchedule') { responseData = saveSchedule(requestData); } else if (action === 'deleteSchedule') { responseData = deleteSchedule(requestData.id); } else if (action === 'saveRoom') { responseData = saveRoom(requestData); } else if (action === 'deleteRoom') { responseData = deleteRoom(requestData.id); } else if (action === 'saveRunningText') { responseData = saveRunningText(requestData.text); } else { responseData = { success: false, error: "Action tidak dikenal: " + action }; } return ContentService.createTextOutput(JSON.stringify(responseData)) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ success: false, error: err.toString() })).setMimeType(ContentService.MimeType.JSON); } finally { lock.releaseLock(); } } /** * ZERO-CONFIG DATABASE SETUP * Otomatis memeriksa dan membuat Tab Sheet beserta Kolom Header jika belum ada. */ function setupDatabase() { const ss = SpreadsheetApp.getActiveSpreadsheet(); configSheet.appendRow(["running_text", "Harap konfirmasi penggunaan ruangan kepada Petugas Operator sebelum kegiatan dimulai."]); } } function getInitialData() { const ss = SpreadsheetApp.getActiveSpreadsheet(); // Read Rooms using getDisplayValues() to preserve exact string values const roomSheet = ss.getSheetByName("ROOMS"); const roomValues = roomSheet.getDataRange().getDisplayValues(); let rooms = []; for (let i = 1; i < roomValues.length; i++) { if (roomValues[i][0]) { rooms.push({ id: String(roomValues[i][0]), name: String(roomValues[i][1]), floor: String(roomValues[i][2]) }); } } // Read Schedules using getDisplayValues() to prevent 1899 Date objects for time const scheduleSheet = ss.getSheetByName("SCHEDULES"); const scheduleValues = scheduleSheet.getDataRange().getDisplayValues(); let schedules = []; for (let i = 1; i < scheduleValues.length; i++) { if (scheduleValues[i][0]) { let rawDate = scheduleValues[i][5]; } function saveSchedule(data) { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName("SCHEDULES"); const values = sheet.getDataRange().getValues(); let rowIndex = -1; for (let i = 1; i < values.length; i++) { if (String(values[i][0]) === String(data.id)) { rowIndex = i + 1; break; } } if (rowIndex > -1) { sheet.getRange(rowIndex, 1, 1, 6).setValues([[data.id, data.roomId, data.agenda, data.startTime, data.endTime, data.date]]); break; } } return { success: true }; } function saveRoom(data) { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName("ROOMS"); const values = sheet.getDataRange().getValues(); let rowIndex = -1; for (let i = 1; i < values.length; i++) { if (String(values[i][0]) === String(data.id)) { rowIndex = i + 1; break; } } if (rowIndex > -1) { sheet.getRange(rowIndex, 1, 1, 3).setValues([[data.id, data.name, data.floor]]); } else { sheet.appendRow([data.id, data.name, data.floor]); } return { success: true }; } function deleteRoom(id) { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName("ROOMS"); const values = sheet.getDataRange().getValues(); for (let i = 1; i < values.length; i++) { if (String(values[i][0]) === String(id)) { sheet.deleteRow(i + 1); break; } } return { success: true }; } function saveRunningText(text) { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName("CONFIG"); const values = sheet.getDataRange().getValues(); let rowIndex = -1; for (let i = 1; i < values.length; i++) { if (String(values[i][0]) === "running_text") { rowIndex = i + 1; break; } } if (rowIndex > -1) { sheet.getRange(rowIndex, 2).setValue(text); } else { sheet.appendRow(["running_text", text]); } return { success: true }; }
